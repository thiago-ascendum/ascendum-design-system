<Section n={3} title="Do and don't" lead="The four mistakes we see most, and what to do instead.">
  <Cards columns={2} items={[
    { tone: 'do', title: 'Do: bind roles', text: <>Use <code>--on-surface</code> or <code>--on-surface-variant</code> for text.</> },
    { tone: 'avoid', title: 'Don\'t: pick a primitive', text: <>Never use <code>--neutral-750</code> or a hex code in a component.</> },
  ]} />
  <Cards columns={2} items={[
    { tone: 'do', title: 'Do: let the theme switch', text: <>Set <code>data-theme="dark"</code> and every role changes at once.</> },
    { tone: 'avoid', title: 'Don\'t: write a dark override', text: 'A colour written for dark by hand stops following the tokens.' },
  ]} />
  <Cards columns={2} items={[
    { tone: 'do', title: 'Do: use the component', text: 'Import it and set its properties.' },
    { tone: 'avoid', title: 'Don\'t: copy its CSS', text: 'A copy does not get the next fix.' },
  ]} />
  <Cards columns={2} items={[
    { tone: 'do', title: 'Do: grey text on light surfaces', text: <>Grey or status text sits on <code>--surface</code>, <code>--surface-container</code> or <code>--surface-container-low</code>.</> },
    { tone: 'avoid', title: 'Don\'t: grey text on hover or pressed', text: <>On <code>--surface-container-highest</code>, <code>--surface-hover</code> or <code>--surface-pressed</code> it falls under AA. Use <code>--on-surface</code> there.</> },
  ]} />
</Section>

<Section n={5} title="Words we use">
  <Table head={['Word', 'What it means']} rows={[
    ['Role', <>A colour named by its job, like <code>--on-surface</code> or <code>--outline</code>. You use roles.</>],
    ['Primitive', <>A raw value, like <code>--neutral-750</code>. It sits behind a role. You never use it directly.</>],
    ['Mode', 'Light or Dark, Desktop or Mobile. Switching it changes every value at once.'],
    ['Context', <>Default or Inverted. <code>data-context="inverted"</code> flips a section to the dark band.</>],
    ['Variant', 'One state or size of a component, chosen with its properties.'],
    ['Gate', 'The checks that run on every change. A change that fails them is not published.'],
  ]} />
</Section>
